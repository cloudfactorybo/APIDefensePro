# =====================================================================
# DefensePro API -> Elastic (Fleet Custom API / httpjson)
# Pegar en Kibana > Dev Tools > Console.
#
# Antes de ejecutar: reemplaza __TOKEN__ por el Bearer token de la API
# (Ctrl+H en la consola). No guardes este archivo con el token real.
#
# Por cada endpoint hay 3 requests, en este orden:
#   1. PUT _component_template/logs-<dataset>@custom  (evita que los docs
#      terminen en el failure store: event.dataset es constant_keyword)
#   2. PUT _ingest/pipeline/defensepro-<endpoint>
#   3. POST kbn:/api/fleet/package_policies           (la integración)
#
# Campos: los del equipo van a ECS (observer.*, @timestamp); el resto queda
# en defensepro.<sección>.* con los mismos nombres que la respuesta de la API.
# =====================================================================


# ---------------------------------------------------------------------
# 0. Pipelines compartidos (una sola vez)
# ---------------------------------------------------------------------

PUT _ingest/pipeline/defensepro-common
{
  "description": "DefensePro API: parseo y campos del equipo. Lo llaman todos los pipelines defensepro-*",
  "processors": [
    { "json": { "field": "message", "target_field": "_parsed" } },
    { "drop": { "description": "Respuesta sin equipo", "if": "!(ctx._parsed instanceof Map) || ctx._parsed.deviceName == null" } },
    { "date": { "if": "ctx._parsed.collectedAt != null", "field": "_parsed.collectedAt", "formats": ["ISO8601"], "ignore_failure": true } },
    { "set": { "field": "observer.vendor", "value": "Radware" } },
    { "set": { "field": "observer.product", "value": "DefensePro" } },
    { "set": { "field": "observer.type", "value": "ids" } },
    { "rename": { "field": "_parsed.deviceName", "target_field": "observer.name" } },
    { "rename": { "if": "ctx._parsed.deviceIp != null && ctx._parsed.deviceIp != ''", "field": "_parsed.deviceIp", "target_field": "observer.ip" } },
    { "rename": { "field": "_parsed.deviceUrl", "target_field": "defensepro.url", "ignore_missing": true } },
    { "rename": { "field": "_parsed.errors", "target_field": "defensepro.errors", "ignore_missing": true } },
    { "rename": { "field": "_parsed.unsupported", "target_field": "defensepro.unsupported", "ignore_missing": true } },
    { "remove": { "field": ["_parsed.collectedAt", "_parsed.deviceIp"], "ignore_missing": true } },
    { "set": { "field": "defensepro.collection", "value": "ok" } },
    { "set": { "if": "ctx.defensepro.errors instanceof List && ctx.defensepro.errors.size() > 0", "field": "defensepro.collection", "value": "partial" } },
    { "set": { "field": "event.type", "value": ["info"] } }
  ]
}

PUT _ingest/pipeline/defensepro-finalize
{
  "description": "DefensePro API: mueve el resto de la respuesta a defensepro.<sección> y limpia",
  "processors": [
    {
      "script": {
        "lang": "painless",
        "source": """
          if (ctx._dp?.section != null && ctx._parsed instanceof Map) {
            ctx.defensepro[ctx._dp.section] = ctx._parsed;
          }
        """
      }
    },
    { "remove": { "field": ["_parsed", "_dp", "message"], "ignore_missing": true } }
  ]
}

          String key(def s) {

            String t = s.toString().toLowerCase();
            StringBuilder b = new StringBuilder();
            boolean sep = false;
            for (int i = 0; i < t.length(); i++) {

              if (Character.isLetterOrDigit(c)) { b.append(c); sep = false; }
              else if (!sep && b.length() > 0) { b.append('_'); sep = true; }
            }
            String r = b.toString();
            if (r.endsWith('_')) { r = r.substring(0, r.length() - 1); }
            return r.isEmpty() ? 'unknown' : r;
          }
          Map byKey(def list, String field) {
            Map m = new HashMap();
            if (list instanceof List) {
              for (def it : list) {
                if (it instanceof Map) {
                  Map c = new HashMap(it);
                  def k = c.remove(field);
                  m.put(key(k), c);
                }
              }
            }
            return m;
          }
          def p = ctx._parsed;

          if (p.dns instanceof Map) {

            }
